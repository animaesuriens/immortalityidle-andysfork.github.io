```typescript
import { EffectHandler, HandlerRegistry } from './handler.interface';
import { ChanceEffect, Effect } from '../types/effect.types';
import { EffectContext, toFormulaContext } from '../types/context.types';
import { RenderedEffect } from '../types/render.types';
import { evaluateAmount } from '../utils/render-helpers';

/**
 * Late-bound reference to handler registry for nested effect execution.
 * Same pattern as conditional.handler.ts.
 */
let registryRef: HandlerRegistry | null = null;

export function setChanceRegistryRef(registry: HandlerRegistry): void {
  registryRef = registry;
}

function executeEffect(effect: Effect, context: EffectContext): void {
  if (!registryRef) {
    throw new Error('Handler registry not initialized for chance handler');
  }
  const handler = registryRef[effect.kind] as EffectHandler<typeof effect>;
  handler.execute(effect, context);
}

function renderEffect(effect: Effect, context: EffectContext): RenderedEffect | RenderedEffect[] {
  if (!registryRef) {
    throw new Error('Handler registry not initialized for chance handler');
  }
  const handler = registryRef[effect.kind] as EffectHandler<typeof effect>;
  return handler.render(effect, context);
}

function flattenEffects(effect: RenderedEffect | RenderedEffect[]): RenderedEffect[] {
  return Array.isArray(effect) ? effect : [effect];
}

export const chanceHandler: EffectHandler<ChanceEffect> = {
  execute(effect: ChanceEffect, context: EffectContext): void {
    const formulaContext = toFormulaContext(context);
    const probability = evaluateAmount(effect.probability, formulaContext);

    // Roll once per execution
    if (Math.random() < probability) {
      for (const nested of effect.effects) {
        executeEffect(nested, context);
      }
    }
  },

  render(effect: ChanceEffect, context: EffectContext): RenderedEffect[] {
    const formulaContext = toFormulaContext(context);
    const probability = evaluateAmount(effect.probability, formulaContext);
    const percentStr = `${Math.round(probability * 100)}%`;

    // Render nested effects with chance annotation (Decision 22: static evaluation)
    const results: RenderedEffect[] = [];
    for (const nested of effect.effects) {
      const rendered = flattenEffects(renderEffect(nested, context));
      for (const r of rendered) {
        results.push({
          ...r,
          condition: r.condition ? `${percentStr} ${r.condition}` : `${percentStr}`,
        });
      }
    }
    return results;
  },
};
```

3. Implement `item-add.handler.ts`:

```typescript
import { EffectHandler } from './handler.interface';
import { ItemAddEffect } from '../types/effect.types';
import { EffectContext, toFormulaContext } from '../types/context.types';
import { RenderedEffect } from '../types/render.types';
import { evaluateAmount } from '../utils/render-helpers';

export const itemAddHandler: EffectHandler<ItemAddEffect> = {
  execute(effect: ItemAddEffect, context: EffectContext): void {
    // Decision 15: silently fail if no inventory space
    if (!context.hasInventorySlots()) {
      return;
    }

    const formulaContext = toFormulaContext(context);
    const quantity = effect.quantity
      ? Math.floor(evaluateAmount(effect.quantity, formulaContext))
      : 1;

    if (quantity <= 0) {
      return;
    }

    // Simple itemId lookup for now (factory support in 04-04)
    if (effect.itemId) {
      context.addItem(effect.itemId, quantity);
      context.emitEvent({
        kind: 'itemAdded',
        itemId: effect.itemId,
        quantity,
      });
    }
  },

  render(effect: ItemAddEffect, context: EffectContext): RenderedEffect {
    const formulaContext = toFormulaContext(context);
    const quantity = effect.quantity
      ? Math.floor(evaluateAmount(effect.quantity, formulaContext))
      : 1;

    const itemName = formatItemName(effect.itemId ?? 'item');

    return {
      kind: 'item',
      visible: true,
      positive: true,
      short: {
        sign: '+',
        amount: quantity,
        label: itemName,
      },
      long: {
        verb: 'Adds',
        amount: quantity,
        name: itemName.toLowerCase(),
      },
    };
  },
};

function formatItemName(itemId: string): string {
  // Capitalize first letter
  return itemId.charAt(0).toUpperCase() + itemId.slice(1);
}
```

/**
 * Reference follower job power.
 *
 * Evaluates to: context.getFollowerPower(job)
 * Renders to: "Hunter Power", "Builder Power", etc.
 *
 * @param job The follower job to get power for
 * @returns A formula that evaluates to the job's total power
 *
 * @example
 * followerPower('hunter')  // Evaluates to hunter total power
 * floor(div(followerPower('hunter'), 20))  // floor(Hunter Power / 20)
 */
export function followerPower(job: string): Formula {
  return {
    evaluate(context: FormulaContext): number {
      // Access via extended context if available
      // The formula context doesn't have getFollowerPower, so we need to extend FormulaContext
      // or evaluate at handler level with EffectContext
      //
      // WORKAROUND: Store follower power in context.variables during context creation
      return context.variables[`_followerPower_${job}`] ?? 0;
    },
    render(_context: FormulaContext, format: FormulaRenderFormat): string {
      const displayName = `${job.charAt(0).toUpperCase() + job.slice(1)} Power`;
      switch (format) {
        case 'value':
          return String(this.evaluate(_context));
        case 'formula':
          return displayName;
        case 'both':
          return `${displayName} (${this.evaluate(_context)})`;
      }
    },
  };
}
