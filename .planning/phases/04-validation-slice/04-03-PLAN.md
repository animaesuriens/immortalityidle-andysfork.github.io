```typescript
/**
 * Chance effect handler - executes nested effects with probability.
 * Probability can be a number (0-1) or Formula for dynamic calculation.
 */

import { EffectHandler, HandlerRegistry } from './handler.interface';
import { ChanceEffect, Effect } from '../types/effect.types';
import { EffectContext } from '../types/context.types';
import { RenderedEffect } from '../types/render.types';
import { toFormulaContext, evaluateAmount } from '../utils/render-helpers';

// Late-bound registry reference (same pattern as conditional.handler)
let registryRef: HandlerRegistry | null = null;

export function setChanceRegistryRef(registry: HandlerRegistry): void {
  registryRef = registry;
}

function executeNestedEffect(effect: Effect, context: EffectContext): void {
  if (!registryRef) {
    throw new Error('Handler registry not initialized for chance handler');
  }
  const handler = registryRef[effect.kind] as EffectHandler<typeof effect>;
  handler.execute(effect, context);
}

function renderNestedEffect(effect: Effect, context: EffectContext): RenderedEffect | RenderedEffect[] {
  if (!registryRef) {
    throw new Error('Handler registry not initialized for chance handler');
  }
  const handler = registryRef[effect.kind] as EffectHandler<typeof effect>;
  return handler.render(effect, context);
}

function flattenEffects(effect: RenderedEffect | RenderedEffect[]): RenderedEffect[] {
  return Array.isArray(effect) ? effect : [effect];
}

export const chanceHandler: EffectHandler<ChanceEffect> = {
  execute(effect: ChanceEffect, context: EffectContext): void {
    const formulaContext = toFormulaContext(context);
    const probability = evaluateAmount(effect.probability, formulaContext);

    // Roll the dice - single roll per execution
    if (Math.random() >= probability) {
      return; // Failed roll
    }

    // Success - execute nested effects
    for (const nested of effect.effects) {
      try {
        executeNestedEffect(nested, context);
      } catch (error) {
        const errorBehavior = nested.onError ?? 'continue';
        if (errorBehavior === 'abort') throw error;
        console.error(`Nested effect error in chance: ${nested.kind}`, error);
      }
    }
  },

  render(effect: ChanceEffect, context: EffectContext): RenderedEffect[] {
    const formulaContext = toFormulaContext(context);
    const probability = evaluateAmount(effect.probability, formulaContext);
    const percentStr = `${Math.round(probability * 100)}%`;

    // Render nested effects with chance annotation (Decision 22: static evaluation)
    const results: RenderedEffect[] = [];
    for (const nested of effect.effects) {
      const rendered = flattenEffects(renderNestedEffect(nested, context));
      for (const r of rendered) {
        results.push({
          ...r,
          condition: r.condition ? `${percentStr} ${r.condition}` : `${percentStr}`,
        });
      }
    }
    return results;
  },
};
```

```typescript
/**
 * Spawn-enemy effect handler - adds enemies to battle queue.
 * Supports inline config or named enemy reference.
 * Renders as "attract [Enemy] trouble" format (Decision 25).
 */

import { EffectHandler } from './handler.interface';
import { SpawnEnemyEffect, EnemyConfig } from '../types/effect.types';
import { EffectContext } from '../types/context.types';
import { RenderedEffect } from '../types/render.types';

export const spawnEnemyHandler: EffectHandler<SpawnEnemyEffect> = {
  execute(effect: SpawnEnemyEffect, context: EffectContext): void {
    // Get enemy config from inline definition or lookup
    const enemyConfig = effect.enemy ?? lookupEnemy(effect.enemyId);
    if (!enemyConfig) {
      console.warn(`Unknown enemy: ${effect.enemyId}`);
      return;
    }

    context.spawnEnemy(enemyConfig);
    // Event emission happens inside context.spawnEnemy()
  },

  render(effect: SpawnEnemyEffect, context: EffectContext): RenderedEffect {
    const enemyConfig = effect.enemy ?? lookupEnemy(effect.enemyId);
    const enemyName = enemyConfig?.name ?? effect.enemyId ?? 'unknown enemy';

    // Decision 25: "attract [Enemy] trouble" format
    return {
      kind: 'spawn.enemy',
      visible: true,
      positive: false,
      short: {
        sign: '',
        amount: 0,
        label: `attract ${enemyName} trouble`,
      },
      long: {
        verb: 'May attract',
        amount: 0,
        name: `${enemyName} trouble`,
      },
    };
  },
};
```

```typescript
/**
 * Item-add effect handler - adds items to inventory.
 * Supports simple item lookup, quantity formulas, and factory generation.
 * Silently fails if no inventory space (Decision 15).
 */

import { EffectHandler } from './handler.interface';
import { ItemAddEffect } from '../types/effect.types';
import { EffectContext } from '../types/context.types';
import { RenderedEffect } from '../types/render.types';
import { toFormulaContext, evaluateAmount } from '../utils/render-helpers';
import { Formula } from '../types/formula.types';

export const itemAddHandler: EffectHandler<ItemAddEffect> = {
  execute(effect: ItemAddEffect, context: EffectContext): void {
    // Decision 15: silently fail if no inventory space
    if (!context.hasInventorySlots()) {
      return;
    }

    const formulaContext = toFormulaContext(context);

    if (effect.itemId) {
      // Simple item lookup path
      const quantity = effect.quantity
        ? Math.floor(evaluateAmount(effect.quantity, formulaContext))
        : 1;

      if (quantity > 0) {
        context.addItem(effect.itemId, quantity);
        context.emitEvent({ kind: 'itemAdded', itemId: effect.itemId, quantity });
      }
    } else if (effect.factory) {
      // Factory-generated item path (Decision 17)
      const args = resolveFactoryArgs(effect.args ?? [], context, formulaContext);
      callFactory(effect.factory, args, context);
    }
  },

  render(effect: ItemAddEffect, context: EffectContext): RenderedEffect {
    const formulaContext = toFormulaContext(context);

    if (effect.itemId) {
      const quantity = effect.quantity
        ? Math.floor(evaluateAmount(effect.quantity, formulaContext))
        : 1;
      const itemName = formatItemName(effect.itemId);

      return {
        kind: 'item.add',
        visible: true,
        positive: true,
        short: {
          sign: '+',
          amount: quantity,
          label: itemName,
        },
        long: {
          verb: 'Adds',
          amount: quantity,
          name: itemName.toLowerCase(),
        },
        formula: typeof effect.quantity === 'object'
          ? { type: 'formula', expression: (effect.quantity as Formula).render(formulaContext, 'both') }
          : { type: 'fixed', base: quantity },
      };
    } else if (effect.factory) {
      // Decision 24: factory items render with conditional format
      const factoryName = formatFactoryName(effect.factory);
      return {
        kind: 'item.add',
        visible: true,
        positive: true,
        short: {
          sign: '+',
          amount: 1,
          label: factoryName,
        },
        long: {
          verb: 'Creates',
          amount: 1,
          name: factoryName.toLowerCase(),
        },
        condition: 'if materials',
      };
    }

    // Fallback for invalid effect
    return {
      kind: 'item.add',
      visible: false,
      positive: true,
      short: { sign: '', amount: 0, label: '' },
      long: { verb: '', amount: 0, name: '' },
    };
  },
};

function resolveFactoryArgs(
  args: (number | Formula | { ref: 'variable'; name: string })[],
  context: EffectContext,
  formulaContext: ReturnType<typeof toFormulaContext>
): number[] {
  return args.map(arg => {
    if (typeof arg === 'number') return arg;
    if (typeof arg === 'object' && 'ref' in arg && arg.ref === 'variable') {
      return context.variables[arg.name] ?? 0;
    }
    // Formula
    return evaluateAmount(arg as Formula, formulaContext);
  });
}

function callFactory(factory: string, args: number[], context: EffectContext): void {
  switch (factory) {
    case 'generateWeapon':
      context.generateWeapon(args[0] ?? 1, args[1]?.toString() ?? 'metal');
      context.emitEvent({ kind: 'itemAdded', itemId: 'weapon', quantity: 1 });
      break;
    case 'generateArmor':
      context.generateArmor(args[0] ?? 1, 'body' as any);
      context.emitEvent({ kind: 'itemAdded', itemId: 'armor', quantity: 1 });
      break;
    case 'generatePotion':
      context.generatePotion(args[0] ?? 1);
      context.emitEvent({ kind: 'itemAdded', itemId: 'potion', quantity: 1 });
      break;
    case 'generatePill':
      context.generatePill(args[0] ?? 1);
      context.emitEvent({ kind: 'itemAdded', itemId: 'pill', quantity: 1 });
      break;
    default:
      console.warn(`Unknown factory: ${factory}`);
  }
}

function formatItemName(itemId: string): string {
  return itemId.charAt(0).toUpperCase() + itemId.slice(1);
}

function formatFactoryName(factory: string): string {
  const mappings: Record<string, string> = {
    'generateWeapon': 'Weapon',
    'generateArmor': 'Armor',
    'generatePotion': 'Potion',
    'generatePill': 'Pill',
  };
  return mappings[factory] ?? factory;
}
```

```typescript
/**
 * Check if specific furniture exists in workbench slot.
 * Returns 1 if present, 0 if not.
 */
export function hasFurniture(furnitureId: string): Formula {
  return {
    evaluate(context: FormulaContext): number {
      return context.hasFurniture('workbench', furnitureId) ? 1 : 0;
    },
    render(context: FormulaContext, format: RenderFormat): string {
      const has = context.hasFurniture('workbench', furnitureId);
      switch (format) {
        case 'value': return has ? '1' : '0';
        case 'expression': return `has ${furnitureId}`;
        case 'both': return `has ${furnitureId} (${has ? 'yes' : 'no'})`;
      }
    },
  };
}

/**
 * Conditional formula: if condition formula > 0, return thenValue, else elseValue.
 * Useful for furniture bonuses: conditional(hasFurniture('dogKennel'), 0.4, 0)
 */
export function conditional(
  condition: Formula,
  thenValue: Formula | number,
  elseValue: Formula | number
): Formula {
  return {
    evaluate(context: FormulaContext): number {
      const condResult = condition.evaluate(context);
      if (condResult > 0) {
        return typeof thenValue === 'number' ? thenValue : thenValue.evaluate(context);
      }
      return typeof elseValue === 'number' ? elseValue : elseValue.evaluate(context);
    },
    render(context: FormulaContext, format: RenderFormat): string {
      const condResult = condition.evaluate(context);
      const result = condResult > 0
        ? (typeof thenValue === 'number' ? thenValue : thenValue.evaluate(context))
        : (typeof elseValue === 'number' ? elseValue : elseValue.evaluate(context));

      switch (format) {
        case 'value': return String(result);
        case 'expression': return `conditional`;
        case 'both': return `${result}`;
      }
    },
  };
}
```
