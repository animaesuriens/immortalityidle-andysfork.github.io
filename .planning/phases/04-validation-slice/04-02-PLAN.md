```typescript
import { EffectHandler } from './handler.interface';
import { ProgressEffect } from '../types/effect.types';
import { EffectContext, toFormulaContext } from '../types/context.types';
import { RenderedEffect, FormulaBreakdown } from '../types/render.types';
import { evaluateAmount } from '../utils/render-helpers';

/**
 * Handler for ProgressEffect.
 * Increments progress counters for impossible tasks.
 */
export const progressHandler: EffectHandler<ProgressEffect> = {
  execute(effect: ProgressEffect, context: EffectContext): void {
    const formulaContext = toFormulaContext(context);
    const amount = effect.amount
      ? evaluateAmount(effect.amount, formulaContext)
      : 1;
    context.incrementProgress(effect.progressType, amount);
    context.checkProgressCompletion();
  },

  render(effect: ProgressEffect, context: EffectContext): RenderedEffect {
    const formulaContext = toFormulaContext(context);
    const amount = effect.amount
      ? evaluateAmount(effect.amount, formulaContext)
      : 1;

    // Build formula breakdown
    let formula: FormulaBreakdown;
    if (effect.amount === undefined || typeof effect.amount === 'number') {
      formula = { type: 'fixed', base: amount };
    } else {
      formula = {
        type: 'fixed',
        expression: effect.amount.render(formulaContext, 'both'),
      };
    }

    // Format progress type for display
    const displayName = formatProgressType(effect.progressType);

    return {
      kind: 'progress',
      visible: true,
      positive: true,
      short: {
        sign: '+',
        amount: Math.floor(amount),
        label: `${displayName} Progress`,
      },
      long: {
        verb: 'Adds',
        amount: Math.floor(amount),
        name: `${displayName.toLowerCase()} progress`,
      },
      formula,
    };
  },
};

/**
 * Format progress type string for display.
 * 'BuildTower' -> 'Tower', 'RaiseIsland' -> 'Island'
 */
function formatProgressType(progressType: string): string {
  // Remove common prefixes and return meaningful name
  const mappings: Record<string, string> = {
    'BuildTower': 'Tower',
    'RaiseIsland': 'Island',
    'TameWinds': 'Wind',
    'SwimOcean': 'Ocean Swim',
    'WriteBook': 'Book',
  };
  return mappings[progressType] ?? progressType;
}
```

In `activity.service.ts`, convert BuildTower to DeclarativeActivity.

BuildTower is complex because it has multiple prerequisite checks (builders, scaffolding, mortar, bricks). The current implementation uses early returns with health damage for failures.

Strategy: Use sequential conditional effects. Each check that fails prevents subsequent effects and causes damage.

```typescript
this.BuildTower = {
  level: 0,
  name: ['Build the Next Level'],
  imageBaseName: 'buildtower',
  activityType: ActivityType.BuildTower,
  duration: 1,
  description: [...],
  consequenceDescription: [...],
  effectsLegacy: ['+Tower Level (requires 10 Builders, Scaffolding, 100 Mortar, 1000 Bricks)'],
  effects: {
    0: [
      // Stamina cost always applies
      { kind: 'status', status: 'stamina', amount: -1000 },

      // Builder check: if not enough builders, damage and stop
      {
        kind: 'conditional',
        condition: {
          kind: 'CompareProperty',
          path: 'followerCount.builder',
          operator: '<',
          value: 10,
        },
        then: [
          // Not enough builders - take 5% max health damage
          // Note: This requires statusMax formula or modifyStatus with formula
          // For now, log injury message
          { kind: 'status', status: 'health', amount: mult(-0.05, statusMax('health')) },
        ],
        else: [
          // Have builders, check scaffolding
          {
            kind: 'conditional',
            condition: { kind: 'HasInventory', check: 'hasItem', itemId: 'scaffolding', quantity: 1 },
            then: [
              { kind: 'item.consume', itemType: 'scaffolding' },
              // Check mortar
              {
                kind: 'conditional',
                condition: { kind: 'HasInventory', check: 'hasItem', itemId: 'mortar', quantity: 100 },
                then: [
                  { kind: 'item.consume', itemType: 'mortar', quantity: 100 },
                  // Check bricks
                  {
                    kind: 'conditional',
                    condition: { kind: 'HasInventory', check: 'hasItem', itemId: 'brick', quantity: 1000 },
                    then: [
                      { kind: 'item.consume', itemType: 'brick', quantity: 1000 },
                      // All checks passed - increment progress
                      { kind: 'progress', progressType: 'BuildTower', amount: 1 },
                    ],
                    else: [
                      // Not enough bricks - 20% damage
                      { kind: 'status', status: 'health', amount: mult(-0.2, statusMax('health')) },
                    ],
                  },
                ],
                else: [
                  // Not enough mortar - 20% damage
                  { kind: 'status', status: 'health', amount: mult(-0.2, statusMax('health')) },
                ],
              },
            ],
            else: [
              // No scaffolding - 20% damage
              { kind: 'status', status: 'health', amount: mult(-0.2, statusMax('health')) },
            ],
          },
        ],
      },
    ],
  },
  consequence: [...], // Keep legacy
  resourceUse: [...],
  requirements: [{}],
  unlocked: true,
  skipApprenticeshipLevel: 0,
} as Activity;
```

**Note:** This nested conditional structure matches the original imperative logic. The HasInventory condition with quantity parameter may need to be extended in condition-evaluator.ts if not already supported.

**Simplification option:** If the nested structure is too complex, consider keeping BuildTower as legacy for now and just implementing the progress handler for testing with a simpler activity. User decision in CONTEXT.md specifies BuildTower, so implement fully.

**Alternative:** Create a helper function that wraps the common "check-consume-or-damage" pattern.
