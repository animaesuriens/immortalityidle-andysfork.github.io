/**
 * Progress effect handler - increments task progress counters.
 * Integrates with ImpossibleTaskService for tower building, etc.
 */

import { EffectHandler } from './handler.interface';
import { ProgressEffect } from '../types/effect.types';
import { EffectContext } from '../types/context.types';
import { RenderedEffect, FormulaBreakdown } from '../types/render.types';
import { toFormulaContext, evaluateAmount } from '../utils/render-helpers';

export const progressHandler: EffectHandler<ProgressEffect> = {
  execute(effect: ProgressEffect, context: EffectContext): void {
    const formulaContext = toFormulaContext(context);
    const amount = effect.amount
      ? evaluateAmount(effect.amount, formulaContext)
      : 1;

    context.incrementProgress(effect.progressType, Math.floor(amount));
    context.checkProgressCompletion();
  },

  render(effect: ProgressEffect, context: EffectContext): RenderedEffect {
    const formulaContext = toFormulaContext(context);
    const amount = effect.amount
      ? evaluateAmount(effect.amount, formulaContext)
      : 1;
    const rounded = Math.floor(amount);

    const displayName = formatProgressType(effect.progressType);

    let formula: FormulaBreakdown;
    if (effect.amount === undefined || typeof effect.amount === 'number') {
      formula = { type: 'fixed', base: rounded };
    } else {
      formula = {
        type: 'formula',
        expression: effect.amount.render(formulaContext, 'both'),
      };
    }

    return {
      kind: 'progress',
      visible: true,
      positive: true,
      short: {
        sign: '+',
        amount: rounded,
        label: `${displayName} Progress`,
      },
      long: {
        verb: 'Adds',
        amount: rounded,
        name: `${displayName.toLowerCase()} progress`,
      },
      formula,
    };
  },
};

function formatProgressType(progressType: string): string {
  const mappings: Record<string, string> = {
    'BuildTower': 'Tower',
    'RaiseIsland': 'Island',
    'TameWinds': 'Wind',
    'LearnToFly': 'Flight',
  };
  return mappings[progressType] ?? progressType;
}
```

/**
 * Item-consume effect handler - removes items from inventory.
 * Can optionally store the consumed item's grade in a context variable.
 * Default error behavior: 'abort' (don't continue if item not found).
 */

import { EffectHandler } from './handler.interface';
import { ItemConsumeEffect } from '../types/effect.types';
import { EffectContext } from '../types/context.types';
import { RenderedEffect } from '../types/render.types';

export const itemConsumeHandler: EffectHandler<ItemConsumeEffect> = {
  execute(effect: ItemConsumeEffect, context: EffectContext): void {
    const quantity = effect.quantity ?? 1;
    const minGrade = effect.minGrade ?? 0;

    // consumeItem returns the grade of consumed item, or 0 if none found
    const grade = context.consumeItem(effect.itemType, minGrade);

    if (grade === 0 && minGrade > 0) {
      // Item not found or didn't meet grade requirement
      // Default behavior for item.consume is 'abort' - throw to stop effect chain
      const errorBehavior = effect.onError ?? 'abort';
      if (errorBehavior === 'abort') {
        throw new Error(`Item consumption failed: ${effect.itemType} (min grade ${minGrade})`);
      }
      // 'continue' or 'skip' - just return without storing grade
      return;
    }

    // Store grade if requested
    if (effect.storeGradeAs) {
      context.setVariable(effect.storeGradeAs, grade);
    }

    // Emit event
    context.emitEvent({
      kind: 'itemConsumed',
      itemType: effect.itemType,
      grade,
    });
  },

  render(effect: ItemConsumeEffect, context: EffectContext): RenderedEffect {
    const quantity = effect.quantity ?? 1;
    const itemName = formatItemType(effect.itemType);

    return {
      kind: 'item.consume',
      visible: true,
      positive: false, // Consumption is a cost
      short: {
        sign: '-',
        amount: quantity,
        label: itemName,
      },
      long: {
        verb: 'Consumes',
        amount: quantity,
        name: itemName.toLowerCase(),
      },
      formula: { type: 'fixed', base: quantity },
    };
  },
};

function formatItemType(itemType: string): string {
  // Capitalize first letter
  return itemType.charAt(0).toUpperCase() + itemType.slice(1);
}
```

function evaluateHasInventory(condition: HasInventory, context: EffectContext): boolean {
  if (condition.check === 'hasSlots') {
    return context.hasInventorySlots();
