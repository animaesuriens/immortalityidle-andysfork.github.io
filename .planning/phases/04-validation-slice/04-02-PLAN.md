   incrementProgress(progressType: string, amount = 1): void {
     const taskType = this.resolveProgressType(progressType);
     if (taskType !== null && this.impossibleTaskService.taskProgress[taskType]) {
       this.impossibleTaskService.taskProgress[taskType].progress += amount;
       this.emitEvent({ kind: 'progressUpdated', progressType, amount });
     }
   }

   private resolveProgressType(progressType: string): ImpossibleTaskType | null {
     const mapping: Record<string, ImpossibleTaskType> = {
       'BuildTower': ImpossibleTaskType.BuildTower,
       'RaiseIsland': ImpossibleTaskType.RaiseIsland,
       'TameWinds': ImpossibleTaskType.TameWinds,
     };
     return mapping[progressType] ?? null;
   }

   checkProgressCompletion(): void {
     this.impossibleTaskService.checkCompletion();
   }

Implement `progress.handler.ts` following the money.handler.ts pattern:

```typescript
import { EffectHandler } from './handler.interface';
import { ProgressEffect } from '../types/effect.types';
import { EffectContext, toFormulaContext } from '../types/context.types';
import { RenderedEffect, FormulaBreakdown } from '../types/render.types';
import { evaluateAmount } from '../utils/render-helpers';

export const progressHandler: EffectHandler<ProgressEffect> = {
  execute(effect: ProgressEffect, context: EffectContext): void {
    const formulaContext = toFormulaContext(context);
    const amount = effect.amount
      ? evaluateAmount(effect.amount, formulaContext)
      : 1;
    context.incrementProgress(effect.progressType, amount);
    context.checkProgressCompletion();
  },

  render(effect: ProgressEffect, context: EffectContext): RenderedEffect {
    const formulaContext = toFormulaContext(context);
    const amount = effect.amount
      ? evaluateAmount(effect.amount, formulaContext)
      : 1;

    let formula: FormulaBreakdown;
    if (effect.amount === undefined || typeof effect.amount === 'number') {
      formula = { type: 'fixed', base: amount };
    } else {
      formula = {
        type: 'fixed',
        expression: effect.amount.render(formulaContext, 'both'),
      };
    }

    const displayName = formatProgressType(effect.progressType);

    return {
      kind: 'progress',
      visible: true,
      positive: true,
      short: {
        sign: '+',
        amount: Math.floor(amount),
        label: `${displayName} Progress`,
      },
      long: {
        verb: 'Adds',
        amount: Math.floor(amount),
        name: `${displayName.toLowerCase()} progress`,
      },
      formula,
    };
  },
};

function formatProgressType(progressType: string): string {
  const mappings: Record<string, string> = {
    'BuildTower': 'Tower',
    'RaiseIsland': 'Island',
    'TameWinds': 'Wind',
  };
  return mappings[progressType] ?? progressType;
}
```

In `activity.service.ts`, convert BuildTower to DeclarativeActivity.

BuildTower has complex prerequisite checks. Use a FLAT sequential approach with individual conditionals rather than deeply nested structure:

```typescript
import { mult, statusMax } from '../effects';

this.BuildTower = {
  level: 0,
  name: ['Build the Next Level'],
  imageBaseName: 'buildtower',
  activityType: ActivityType.BuildTower,
  duration: 1,
  description: [...],
  consequenceDescription: [...],
  effectsLegacy: ['+Tower Level (requires 10 Builders, Scaffolding, 100 Mortar, 1000 Bricks)'],
  effects: {
    0: [
      // Stamina cost always applies
      { kind: 'status', status: 'stamina', amount: -1000 },

      // Check 1: Not enough builders -> 5% max health damage, skip rest
      {
        kind: 'conditional',
        condition: {
          kind: 'CompareProperty',
          path: 'followerCount.builder',
          operator: '<',
          value: 10,
        },
        then: [
          { kind: 'status', status: 'health', amount: mult(-0.05, statusMax('health')) },
        ],
      },

      // Check 2: Has builders but no scaffolding -> 20% damage, skip rest
      {
        kind: 'conditional',
        condition: {
          kind: 'And',
          conditions: [
            { kind: 'CompareProperty', path: 'followerCount.builder', operator: '>=', value: 10 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'scaffolding', quantity: 1, negate: true },
          ],
        },
        then: [
          { kind: 'status', status: 'health', amount: mult(-0.2, statusMax('health')) },
        ],
      },

      // Check 3: Has builders, scaffolding but no mortar -> consume scaffolding, 20% damage
      {
        kind: 'conditional',
        condition: {
          kind: 'And',
          conditions: [
            { kind: 'CompareProperty', path: 'followerCount.builder', operator: '>=', value: 10 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'scaffolding', quantity: 1 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'mortar', quantity: 100, negate: true },
          ],
        },
        then: [
          { kind: 'item.consume', itemType: 'scaffolding' },
          { kind: 'status', status: 'health', amount: mult(-0.2, statusMax('health')) },
        ],
      },

      // Check 4: Has builders, scaffolding, mortar but no bricks -> consume scaffolding+mortar, 20% damage
      {
        kind: 'conditional',
        condition: {
          kind: 'And',
          conditions: [
            { kind: 'CompareProperty', path: 'followerCount.builder', operator: '>=', value: 10 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'scaffolding', quantity: 1 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'mortar', quantity: 100 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'brick', quantity: 1000, negate: true },
          ],
        },
        then: [
          { kind: 'item.consume', itemType: 'scaffolding' },
          { kind: 'item.consume', itemType: 'mortar', quantity: 100 },
          { kind: 'status', status: 'health', amount: mult(-0.2, statusMax('health')) },
        ],
      },

      // Success path: All requirements met -> consume all, increment progress
      {
        kind: 'conditional',
        condition: {
          kind: 'And',
          conditions: [
            { kind: 'CompareProperty', path: 'followerCount.builder', operator: '>=', value: 10 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'scaffolding', quantity: 1 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'mortar', quantity: 100 },
            { kind: 'HasInventory', check: 'hasItem', itemId: 'brick', quantity: 1000 },
          ],
        },
        then: [
          { kind: 'item.consume', itemType: 'scaffolding' },
          { kind: 'item.consume', itemType: 'mortar', quantity: 100 },
          { kind: 'item.consume', itemType: 'brick', quantity: 1000 },
          { kind: 'progress', progressType: 'BuildTower', amount: 1 },
        ],
      },
    ],
  },
  consequence: [...], // Keep legacy
  resourceUse: [...],
  requirements: [{}],
  unlocked: true,
  skipApprenticeshipLevel: 0,
} as Activity;
```

**Note:** This flat sequential structure means each condition is checked independently. Only ONE branch will have its condition be true at any time (they are mutually exclusive by design), so effects won't double-apply.

The `statusMax('health')` formula is already implemented in formula.builders.ts.
