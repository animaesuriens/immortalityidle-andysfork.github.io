 * Add item to inventory effect.
 * Supports both direct itemId lookup and factory-generated items.
 */
export interface ItemAddEffect extends BaseEffect {
  readonly kind: 'item.add';
  /** Direct item ID for simple lookup */
  readonly itemId?: string;
  /** Factory function name for generated items */
  readonly factory?: 'generateWeapon' | 'generateArmor' | 'generatePotion' | 'generatePill';
  /** Arguments for factory function - can include variable references */
  readonly args?: readonly FactoryArg[];
  /** Quantity (for itemId mode) */
  readonly quantity?: number | Formula;
}

/**
 * Factory argument - can be a number, formula, or variable reference.
 */
export type FactoryArg = number | Formula | VariableRef;

/**
 * Variable reference for factory arguments.
 * Evaluates to the value of a context variable at execution time.
 */
export interface VariableRef {
  readonly ref: 'variable';
  readonly name: string;
}

import { EffectHandler } from './handler.interface';
import { ItemConsumeEffect } from '../types/effect.types';
import { EffectContext, toFormulaContext } from '../types/context.types';
import { RenderedEffect } from '../types/render.types';

/**
 * Handler for ItemConsumeEffect.
 * Consumes an item by type and optionally stores its grade in a variable.
 *
 * Decision 5: consume returns grade
 * Decision 9: storeGradeAs for variable capture
 * Decision 20: item-consume defaults to 'abort' on failure
 */
export const itemConsumeHandler: EffectHandler<ItemConsumeEffect> = {
  execute(effect: ItemConsumeEffect, context: EffectContext): void {
    const grade = context.consumeItem(effect.itemType, effect.minGrade);

    // If consumption failed (no item found), grade is 0
    if (grade < (effect.minGrade ?? 1)) {
      // Default behavior: abort effect chain
      // Decision 20: item-consume aborts by default
      const onError = effect.onError ?? 'abort';
      if (onError === 'abort') {
        throw new Error(`Item consume failed: no ${effect.itemType} with grade >= ${effect.minGrade ?? 1}`);
      }
      // 'continue' or 'skip' - just don't store anything
      return;
    }

    // Store grade in variable for subsequent effects (Decision 9)
    if (effect.storeGradeAs) {
      context.setVariable(effect.storeGradeAs, grade);
    }

    context.emitEvent({
      kind: 'itemConsumed',
      itemType: effect.itemType,
      grade,
    });
  },

  render(effect: ItemConsumeEffect, _context: EffectContext): RenderedEffect {
    const itemName = formatItemType(effect.itemType);

    // Decision 23: render with negative prefix and red coloring
    // The 'positive: false' flag signals red coloring
    return {
      kind: 'item',
      visible: true,
      positive: false, // Triggers red coloring
      short: {
        sign: '-',
        amount: 1,
        label: itemName,
      },
      long: {
        verb: 'Consumes',
        amount: 1,
        name: itemName.toLowerCase(),
      },
    };
  },
};

function formatItemType(itemType: string): string {
  // Capitalize and format item types
  const mappings: Record<string, string> = {
    'metal': 'Metal',
    'wood': 'Wood',
    'hide': 'Hide',
    'ore': 'Ore',
    'scaffolding': 'Scaffolding',
    'mortar': 'Mortar',
    'brick': 'Bricks',
  };
  return mappings[itemType] ?? itemType.charAt(0).toUpperCase() + itemType.slice(1);
}

```typescript
import { EffectHandler } from './handler.interface';
import { ItemAddEffect, FactoryArg, VariableRef } from '../types/effect.types';
import { EffectContext, toFormulaContext } from '../types/context.types';
import { RenderedEffect, FormulaBreakdown } from '../types/render.types';
import { evaluateAmount } from '../utils/render-helpers';
import { Formula } from '../types/formula.types';

/**
 * Type guard for VariableRef.
 */
function isVariableRef(arg: FactoryArg): arg is VariableRef {
  return typeof arg === 'object' && 'ref' in arg && arg.ref === 'variable';
}

/**
 * Resolve a factory argument to a number.
 */
function resolveArg(arg: FactoryArg, context: EffectContext): number {
  if (typeof arg === 'number') {
    return arg;
  }
  if (isVariableRef(arg)) {
    return context.variables[arg.name] ?? 0;
  }
  // It's a Formula
  const formulaContext = toFormulaContext(context);
  return (arg as Formula).evaluate(formulaContext);
}

export const itemAddHandler: EffectHandler<ItemAddEffect> = {
  execute(effect: ItemAddEffect, context: EffectContext): void {
    // Decision 15: silently fail if no inventory space
    if (!context.hasInventorySlots()) {
      return;
    }

    // Factory mode: generate item using context method
    if (effect.factory) {
      const resolvedArgs = effect.args?.map(arg => resolveArg(arg, context)) ?? [];

      switch (effect.factory) {
        case 'generateWeapon': {
          const [grade, materialIndex] = resolvedArgs;
          // Material is passed as string in effect definition, not in args
          // We need to handle this differently - material should be in effect
          const weapon = context.generateWeapon(grade ?? 1, 'metal');
          if (weapon) {
            context.emitEvent({ kind: 'itemAdded', itemId: weapon.id, quantity: 1 });
          }
          break;
        }
        case 'generateArmor': {
          const [grade] = resolvedArgs;
          const armor = context.generateArmor(grade ?? 1, 'body');
          if (armor) {
            context.emitEvent({ kind: 'itemAdded', itemId: armor.id, quantity: 1 });
          }
          break;
        }
        case 'generatePotion': {
          const [grade] = resolvedArgs;
          context.generatePotion(grade ?? 1);
          context.emitEvent({ kind: 'itemAdded', itemId: 'potion', quantity: 1 });
          break;
        }
        case 'generatePill': {
          const [grade] = resolvedArgs;
          context.generatePill(grade ?? 1);
          context.emitEvent({ kind: 'itemAdded', itemId: 'pill', quantity: 1 });
          break;
        }
      }
      return;
    }

    // Simple itemId mode
    if (effect.itemId) {
      const formulaContext = toFormulaContext(context);
      const quantity = effect.quantity
        ? Math.floor(evaluateAmount(effect.quantity, formulaContext))
        : 1;

      if (quantity <= 0) {
        return;
      }

      context.addItem(effect.itemId, quantity);
      context.emitEvent({
        kind: 'itemAdded',
        itemId: effect.itemId,
        quantity,
      });
    }
  },

  render(effect: ItemAddEffect, context: EffectContext): RenderedEffect {
    // Factory mode rendering (Decision 24: conditional format)
    if (effect.factory) {
      const factoryName = formatFactoryName(effect.factory);
      // Show dependency on consumed item
      return {
        kind: 'item',
        visible: true,
        positive: true,
        short: {
          sign: '+',
          amount: 1,
          label: factoryName,
        },
        long: {
          verb: 'Creates',
          amount: 1,
          name: factoryName.toLowerCase(),
        },
        condition: '(if metal)', // Decision 24: show dependency
      };
    }

    // Simple itemId mode
    const formulaContext = toFormulaContext(context);
    const quantity = effect.quantity
      ? Math.floor(evaluateAmount(effect.quantity, formulaContext))
      : 1;
    const itemName = formatItemName(effect.itemId ?? 'item');

    return {
      kind: 'item',
      visible: true,
      positive: true,
      short: {
        sign: '+',
        amount: quantity,
        label: itemName,
      },
      long: {
        verb: 'Adds',
        amount: quantity,
        name: itemName.toLowerCase(),
      },
    };
  },
};

function formatItemName(itemId: string): string {
  return itemId.charAt(0).toUpperCase() + itemId.slice(1);
}

function formatFactoryName(factory: string): string {
  const mappings: Record<string, string> = {
    'generateWeapon': 'Weapon',
    'generateArmor': 'Armor',
    'generatePotion': 'Potion',
    'generatePill': 'Pill',
  };
  return mappings[factory] ?? factory;
}
```
